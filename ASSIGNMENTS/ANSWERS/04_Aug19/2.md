# Assignment 01 — Employee Management System

## QNO 1: Employee Management System

## 1. Display All Employees

```sql
SELECT * FROM employees;
```

---

# SQL Queries

## 2. Salary Greater Than ₹60,000

```sql
SELECT * FROM employees
WHERE salary > 60000;
```

## 3. Salary Less Than ₹50,000

```sql
SELECT * FROM employees
WHERE salary < 50000;
```

## 4. Age Greater Than 30

```sql
SELECT * FROM employees
WHERE age > 30;
```

## 5. Experience Greater Than or Equal to 5 Years

```sql
SELECT * FROM employees
WHERE experience >= 5;
```

## 6. Employees Working in IT or Finance

```sql
SELECT * FROM employees
WHERE department = "IT"
   OR department = "Finance";
```

## 7. Employees Working in IT, HR, or Sales

```sql
SELECT * FROM employees
WHERE department IN ("IT", "HR", "Sales");
```

## 8. Employees Not Working in IT or HR

```sql
SELECT * FROM employees
WHERE department NOT IN ("IT", "HR");
```

## 9. Employees from Indore or Pune

```sql
SELECT * FROM employees
WHERE city IN ("Indore", "Pune");
```

## 10. Employees Not from Indore, Mumbai, or Bhopal

```sql
SELECT * FROM employees
WHERE city NOT IN ("Indore", "Mumbai", "Bhopal");
```

## 11. Salary Between ₹40,000 and ₹70,000

```sql
SELECT * FROM employees
WHERE salary BETWEEN 40000 AND 70000;
```

## 12. Age Between 25 and 35

```sql
SELECT * FROM employees
WHERE age BETWEEN 25 AND 35;
```

## 13. Experience Between 3 and 7 Years

```sql
SELECT * FROM employees
WHERE experience BETWEEN 3 AND 7;
```

## 14. Employees Who Joined Between 2022 and 2025

```sql
SELECT * FROM employees
WHERE joining_date BETWEEN "2022-01-01" AND "2025-12-31";
```

## 15. Employees Who Joined Before 2023

```sql
SELECT * FROM employees
WHERE joining_date < "2023-01-01";
```

## 16. Employees Who Joined After 2024

```sql
SELECT * FROM employees
WHERE joining_date > "2024-12-31";
```

## 17. Employees from Indore AND Working in IT

```sql
SELECT * FROM employees
WHERE city = "Indore"
  AND department = "IT";
```

## 18. Employees Working in IT AND Salary Greater Than ₹70,000

```sql
SELECT * FROM employees
WHERE department = "IT"
  AND salary > 70000;
```

## 19. Salary Between ₹50,000 and ₹90,000 AND Experience Greater Than 4 Years

```sql
SELECT * FROM employees
WHERE salary BETWEEN 50000 AND 90000
  AND experience > 4;
```

## 20. Working in IT or Finance AND Salary Greater Than ₹60,000

```sql
SELECT * FROM employees
WHERE department IN ("IT", "Finance")
  AND salary > 60000;
```

## 21. Joined Between 2021 and 2024 AND Salary Greater Than ₹50,000

```sql
SELECT * FROM employees
WHERE joining_date BETWEEN "2021-01-01" AND "2024-12-31"
  AND salary > 50000;
```

## 22. Salary NOT Between ₹40,000 and ₹70,000

```sql
SELECT * FROM employees
WHERE salary NOT BETWEEN 40000 AND 70000;
```

## 23. Department Not IT, HR, or Sales

```sql
SELECT * FROM employees
WHERE department NOT IN ("IT", "HR", "Sales");
```

## 24. Joining Date NOT Between 2022 and 2025

```sql
SELECT * FROM employees
WHERE joining_date NOT BETWEEN "2022-01-01" AND "2025-12-31";
```

## 25. From Indore or Pune, Salary ₹45,000–₹80,000, Experience More Than 3 Years

```sql
SELECT * FROM employees
WHERE city IN ("Indore", "Pune")
  AND salary BETWEEN 45000 AND 80000
  AND experience > 3;
```

## 26. Did Not Join Between 2020 and 2023 AND Salary Greater Than ₹60,000

```sql
SELECT * FROM employees
WHERE joining_date NOT BETWEEN "2020-01-01" AND "2023-12-31"
  AND salary > 60000;
```

## 27. From Indore, Pune, or Bangalore, Salary ₹50,000–₹90,000, Experience at Least 4 Years

```sql
SELECT * FROM employees
WHERE city IN ("Indore", "Pune", "Bangalore")
  AND salary BETWEEN 50000 AND 90000
  AND experience >= 4;
```

## 28. Not from Bhopal or Mumbai, Not Working in HR, Joined Between 2021 and 2025

```sql
SELECT * FROM employees
WHERE city NOT IN ("Bhopal", "Mumbai")
  AND department != "HR"
  AND joining_date BETWEEN "2021-01-01" AND "2025-12-31";
```

## 29. Working in IT or Finance, Salary Greater Than ₹60,000, Age 25–35, Joined After 2022

```sql
SELECT * FROM employees
WHERE department IN ("IT", "Finance")
  AND salary > 60000
  AND age BETWEEN 25 AND 35
  AND joining_date > "2022-12-31";
```

## 30. Salary NOT ₹40,000–₹75,000, Experience 2–8 Years, Not Working in Sales

```sql
SELECT * FROM employees
WHERE salary NOT BETWEEN 40000 AND 75000
  AND experience BETWEEN 2 AND 8
  AND department != "Sales";
```

## 31. From Indore or Pune, IT/Finance/HR, Salary ₹45,000–₹85,000, Joined 2020–2024

```sql
SELECT * FROM employees
WHERE city IN ("Indore", "Pune")
  AND department IN ("IT", "Finance", "HR")
  AND salary BETWEEN 45000 AND 85000
  AND joining_date BETWEEN "2020-01-01" AND "2024-12-31";
```
