### QNO 1: Matrix Transpose & Diagonal Transformation

```python
def main():
    n = int(input("Enter the size of matrix: "))
    print("\nEnter matrix elements:")
    matrix = []
    for _ in range(n):
        row = list(map(int, input().split()))
        matrix.append(row)

    print("\nOriginal Matrix:")
    for row in matrix:
        print(*row)
    
    print("\nMain Diagonal Elements:")
    main_diag = []
    for i in range(n):
        main_diag.append(matrix[i][i])
    print(*main_diag)
    
    print("\nSecondary Diagonal Elements:")
    sec_diag = []
    for i in range(n):
        sec_diag.append(matrix[i][n-1-i])
    print(*sec_diag)

    # Transpose
    transpose = []
    for i in range(n):
        new_row = []
        for j in range(n):
            new_row.append(matrix[j][i])
        transpose.append(new_row)
        
    print("\nTranspose Matrix:")
    for row in transpose:
        print(*row)
        
    for i in range(n):
        transpose[i][i], transpose[i][n-1-i] = transpose[i][n-1-i], transpose[i][i]
        
    print("\nFinal Matrix After Diagonal Swapping:")
    for row in transpose:
        print(*row)

if __name__ == "__main__":
    main()
```

### QNO 2: Find the Longest Substring That Appears at Both Ends

```python
def find_longest_prefix_suffix():
    s = input().strip()
    n = len(s)
    if n == 0:
        print("Not Found")
        return
        
    lps = [0] * n
    length = 0
    i = 1
    
    while i < n:
        if s[i] == s[length]:
            length += 1
            lps[i] = length
            i += 1
        else:
            if length != 0:
                length = lps[length - 1]
            else:
                lps[i] = 0
                i += 1
                
    longest = lps[-1]
    if longest > 0:
        print(s[:longest])
    else:
        print("Not Found")

if __name__ == "__main__":
    find_longest_prefix_suffix()
```

### QUESTION 3: LIBRARY BOOK RECORD Analysis

```python
from collections import namedtuple

Book = namedtuple('Book', ['book_id', 'title', 'author', 'price'])

def main():
    n = int(input("Enter number of books: "))
    books = []
    for i in range(1, n + 1):
        print(f"\nEnter details for Book {i}:")
        b_id = input("Enter Book ID: ")
        title = input("Enter Title: ")
        author = input("Enter Author: ")
        price = float(input("Enter Price: "))
        books.append(Book(b_id, title, author, price))
        
    print("\nAll Book Details:")
    for b in books:
        price_val = int(b.price) if b.price.is_integer() else b.price
        print(f"{b.book_id} {b.title} {b.author} {price_val}")
        
    most_expensive = max(books, key=lambda b: b.price)
    cheapest = min(books, key=lambda b: b.price)
    avg_price = sum(b.price for b in books) / n
    
    print("\nMost Expensive Book:")
    p_me = int(most_expensive.price) if most_expensive.price.is_integer() else most_expensive.price
    print(f"{most_expensive.book_id} {most_expensive.title} {most_expensive.author} {p_me}")
    
    print("\nCheapest Book:")
    p_c = int(cheapest.price) if cheapest.price.is_integer() else cheapest.price
    print(f"{cheapest.book_id} {cheapest.title} {cheapest.author} {p_c}")
    
    print(f"\nAverage Book Price:\n{avg_price:.1f}")
    
    print("\nBooks With Price Greater Than Average:")
    for b in books:
        if b.price > avg_price:
            p_b = int(b.price) if b.price.is_integer() else b.price
            print(f"{b.book_id} {b.title} {b.author} {p_b}")

if __name__ == "__main__":
    main()
```
