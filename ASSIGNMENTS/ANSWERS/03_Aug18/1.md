## Part 1 — Create Database

```sql
CREATE DATABASE employee_management;

```

## Part 2 — Create `employees` Table

```sql
CREATE TABLE employees (
    employee_id INT UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    employee_code CHAR(6) NOT NULL,
    employee_name VARCHAR(100) NOT NULL,
    age TINYINT UNSIGNED NOT NULL,
    number_of_children TINYINT UNSIGNED DEFAULT 0,
    experience_months SMALLINT UNSIGNED DEFAULT 0,
    salary DECIMAL(10, 2) NOT NULL,
    performance_score FLOAT,
    productivity_score DOUBLE,
    department_id MEDIUMINT UNSIGNED,
    employee_status ENUM('ACTIVE', 'INACTIVE', 'ON_LEAVE', 'RESIGNED') DEFAULT 'ACTIVE',
    skills SET('JAVA', 'PYTHON', 'MYSQL', 'REACT', 'AWS', 'NODEJS'),
    gender ENUM('Male', 'Female', 'Other'),
    date_of_birth DATE,
    joining_date DATE,
    login_time TIME,
    interview_datetime DATETIME,
    birth_year YEAR,
    is_active BOOLEAN DEFAULT TRUE,
    short_note TINYTEXT,
    employee_description TEXT,
    training_history MEDIUMTEXT,
    employee_profile LONGTEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);
```

## Part 4 — TIMESTAMP Requirement

```sql
-- Check created_at and updated_at
SELECT employee_name, created_at, updated_at FROM employees WHERE employee_id = 1;

-- Update the employee's name
UPDATE employees SET employee_name = 'Amitabh Sharma' WHERE employee_id = 1;

-- Check both timestamps again
SELECT employee_name, created_at, updated_at FROM employees WHERE employee_id = 1;
-- Notice that created_at is unchanged, but updated_at has been updated to the current time.
```

## Part 5 — DDL Operations

```sql
-- ALTER TABLE
ALTER TABLE employees ADD phone_number VARCHAR(15);
ALTER TABLE employees ADD email VARCHAR(100);

ALTER TABLE employees MODIFY employee_name VARCHAR(150);
ALTER TABLE employees MODIFY employee_code CHAR(10);

ALTER TABLE employees RENAME COLUMN short_note TO employee_note;

ALTER TABLE employees DROP COLUMN email;
ALTER TABLE employees DROP COLUMN phone_number;

-- AUTO_INCREMENT using ALTER
CREATE TABLE employee_documents (
    document_id INT,
    document_name VARCHAR(100)
);

-- Note: To set AUTO_INCREMENT, the column must be a key.
ALTER TABLE employee_documents MODIFY document_id INT AUTO_INCREMENT PRIMARY KEY;

-- CREATE TABLE AS SELECT
CREATE TABLE employee_backup AS SELECT * FROM employees;

-- TRUNCATE
CREATE TABLE employee_test (
    id INT,
    name VARCHAR(50)
);

INSERT INTO employee_test VALUES (1, 'Test A'), (2, 'Test B');
SELECT * FROM employee_test;

TRUNCATE TABLE employee_test;
SELECT * FROM employee_test; -- Verify data is gone
SHOW TABLES LIKE 'employee_test'; -- Verify table still exists

-- DROP
DROP TABLE employee_test;
DROP TABLE employee_documents;
SHOW TABLES;
```
