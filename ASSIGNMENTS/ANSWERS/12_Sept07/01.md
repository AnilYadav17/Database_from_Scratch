### 1. Departments and Employees (Referential Integrity Enforcement)

```sql
-- Parent Table: departments
CREATE TABLE IF NOT EXISTS departments (
    dept_id INT PRIMARY KEY,
    dept_name VARCHAR(50) NOT NULL
);

-- Child Table: employees
CREATE TABLE IF NOT EXISTS employees (
    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(50) NOT NULL,
    dept_id INT,
    CONSTRAINT fk_emp_department FOREIGN KEY (dept_id)
```

### 2. Customers and Orders (ON DELETE CASCADE & ON UPDATE CASCADE)

```sql
-- Parent Table: customers
CREATE TABLE IF NOT EXISTS customers (
    customer_id INT PRIMARY KEY,
    customer_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE
);

-- Child Table: orders with cascading actions
CREATE TABLE IF NOT EXISTS orders (
    order_id INT PRIMARY KEY,
    order_date DATE NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    customer_id INT,
    CONSTRAINT fk_orders_customer FOREIGN KEY (customer_id) 
        REFERENCES customers(customer_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- Task a: Insert sample records
INSERT INTO customers (customer_id, customer_name, email) VALUES 
(1, 'Aarav Sharma', 'aarav@example.com'),
(2, 'Bhavna Patel', 'bhavna@example.com'),
(3, 'Chetan Bhagat', 'chetan@example.com');

INSERT INTO orders (order_id, order_date, amount, customer_id) VALUES 
(501, '2026-09-01', 1250.00, 1),
(502, '2026-09-02', 3400.50, 1),
(503, '2026-09-03', 890.00, 2),
(504, '2026-09-04', 2100.00, 3);

-- Task b: Test ON UPDATE CASCADE
-- Updating customer_id 1 to 10 in parent table

-- Verify orders table: order 501 and 502 customer_id is automatically 10

-- Task c: Test ON DELETE CASCADE
-- Deleting customer 2

-- Verify orders table: order 503 is automatically deleted
```

### 3. Healthcare Appointment System (Multiple Foreign Keys)

```sql
-- Parent Table 1: doctors
CREATE TABLE IF NOT EXISTS doctors (
    doctor_id INT PRIMARY KEY,
    doctor_name VARCHAR(50) NOT NULL,
    specialization VARCHAR(50)
);

-- Parent Table 2: patients
CREATE TABLE IF NOT EXISTS patients (
    patient_id INT PRIMARY KEY,
    patient_name VARCHAR(50) NOT NULL,
    age INT
);

-- Child Table: appointments referencing both doctors and patients
CREATE TABLE IF NOT EXISTS appointments (
    appointment_id INT PRIMARY KEY,
```

### 4. College Course Enrollment (ON DELETE SET NULL)

```sql
-- Parent Table: courses
CREATE TABLE IF NOT EXISTS courses (
    course_id INT PRIMARY KEY,
    course_name VARCHAR(50) NOT NULL
);

-- Child Table: enrollments with ON DELETE SET NULL
CREATE TABLE IF NOT EXISTS enrollments (
    enrollment_id INT PRIMARY KEY,
    student_name VARCHAR(50) NOT NULL,
    course_id INT NULL,
    CONSTRAINT fk_enroll_course FOREIGN KEY (course_id) 
        REFERENCES courses(course_id) 
        ON DELETE SET NULL
        ON UPDATE CASCADE
);

-- Task a: Insert courses and enrollments
INSERT INTO courses (course_id, course_name) VALUES 
(101, 'Database Systems'),
(102, 'Operating Systems');

INSERT INTO enrollments (enrollment_id, student_name, course_id) VALUES

(3, 'Charlie', 102);

-- Task b: Delete course 101

-- Verify: student records for Alice and Bob still exist with course_id set to NULL
SELECT * FROM enrollments;
```

### 5. Managing Foreign Keys using ALTER TABLE

```sql
-- Create initial tables without foreign key constraint
CREATE TABLE IF NOT EXISTS projects (
    project_id INT PRIMARY KEY,
    project_name VARCHAR(100) NOT NULL
);

CREATE TABLE IF NOT EXISTS project_tasks (
    task_id INT PRIMARY KEY,
    task_title VARCHAR(100) NOT NULL,
    project_id INT NOT NULL
);

-- 1. Add Foreign Key Constraint via ALTER TABLE
ALTER TABLE project_tasks 
ADD CONSTRAINT fk_task_project 
FOREIGN KEY (project_id) REFERENCES projects(project_id)
ON DELETE CASCADE
ON UPDATE CASCADE;

-- 2. Verify Constraint using SHOW CREATE TABLE
SHOW CREATE TABLE project_tasks;

-- 3. Drop Foreign Key Constraint via ALTER TABLE
ALTER TABLE project_tasks 
DROP FOREIGN KEY fk_task_project;
```
